\section{Introduction}

\begin{frame}{What Will We Learn?}
	\begin{itemize}
		\item \textbf{Sequences} -- automatic running numbers (Student/001, INV/2026/0001, \ldots)
		\item \textbf{XML data importing} -- loading master/config data with modules
		\item \textbf{CSV data importing} -- bulk tabular data via CSV files
		\item \textbf{Demo data} -- sample records for training and testing databases
		\item \textbf{Email templates} -- reusable QWeb mail content for notifications
	\end{itemize}
\end{frame}

\begin{frame}{Why These Topics Matter}
	\begin{itemize}
		\item Real Odoo modules need default data, not only Python models.
		\item Sequences give professional, unique document numbers.
		\item XML/CSV data make installs repeatable.
		\item Demo data helps trainers and developers test quickly.
		\item Email templates standardize communication with users/customers.
	\end{itemize}
\end{frame}
