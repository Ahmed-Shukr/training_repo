\section{Introduction}

\begin{frame}{What Will We Learn?}
	\begin{itemize}
		\item \textbf{Scheduled Actions (\texttt{ir.cron})} -- automatic background jobs
		\item \textbf{QWeb templates} -- XML-based templating for reports and HTML
		\item \textbf{Report actions} -- printable PDF/HTML reports from Odoo
	\end{itemize}
\end{frame}

\begin{frame}{Why Automation and Reports Matter}
	\begin{itemize}
		\item Not every process should wait for a user click.
		\item Cron jobs handle reminders, cleanups, syncs, and nightly computations.
		\item Reports turn records into professional PDFs for business use.
		\item QWeb is the engine behind Odoo report layouts.
	\end{itemize}
\end{frame}

\begin{frame}{Where These Pieces Fit}
	\begin{enumerate}
		\item \textbf{Cron} decides \emph{when} automated Python code runs.
		\item \textbf{QWeb} decides \emph{how} HTML/report content is rendered.
		\item \textbf{Report action} decides \emph{which} records and template produce a printable document.
	\end{enumerate}
\end{frame}
