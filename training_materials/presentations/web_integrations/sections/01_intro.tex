\begin{frame}{Backend Web vs Frontend Web in Odoo}
	\begin{itemize}
		\item \textbf{Backend web}: Python controllers expose HTTP/JSON endpoints.
		\item \textbf{External APIs}: XML-RPC / JSON-RPC let outside systems talk to Odoo.
		\item \textbf{Frontend web}: OWL + JavaScript build interactive UI in the browser.
		\item Together they power websites, portals, custom backends, and integrations.
	\end{itemize}
\end{frame}
