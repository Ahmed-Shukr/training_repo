\section{Chapter Four Overview}

\begin{frame}{What This Chapter Covers}
	\begin{itemize}
		\item Transient models and wizard forms
		\item Abstract models
		\item Python and SQL constraints
		\item Extension and delegation inheritance
		\item XPath and view extension
		\item Smart / stat buttons
		\item Advanced views: Pivot, Kanban, Calendar, Graph, Activity, Gantt, and more
	\end{itemize}
\end{frame}

\begin{frame}{Why These Topics Matter}
	\begin{itemize}
		\item Real Odoo apps are not only form + tree views.
		\item Wizards guide users through business processes.
		\item Inheritance lets you extend Odoo without rewriting core modules.
		\item Advanced views turn records into dashboards, boards, and schedules.
		\item Constraints protect data quality at Python and SQL levels.
	\end{itemize}
\end{frame}

\begin{frame}{Learning Goals}
	\begin{itemize}
		\item Build a wizard with a transient model and form view.
		\item Create reusable abstract models.
		\item Add Python and SQL constraints correctly.
		\item Extend models with \texttt{\_inherit} and delegation (\texttt{\_inherits}).
		\item Extend views with XPath inheritances.
		\item Add smart buttons and advanced view types.
	\end{itemize}
\end{frame}
