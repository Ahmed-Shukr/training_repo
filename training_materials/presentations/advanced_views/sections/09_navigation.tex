\section{Navigation and Actions}

\begin{frame}{How Users Reach Your Views}
	\begin{itemize}
		\item Menus and window actions connect models to screens.
		\item A typical chain is:
		\begin{enumerate}
			\item Menu item
			\item Window action (\texttt{ir.actions.act\_window})
			\item View modes + views
		\end{enumerate}
		\item Smart buttons and wizards also return actions for navigation.
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{Window Action Recap}
\begin{lstlisting}[style=xml]
<record id="action_student" model="ir.actions.act_window">
  <field name="name">Students</field>
  <field name="res_model">student.student</field>
  <field name="view_mode">list,kanban,form,graph,pivot</field>
  <field name="context">{}</field>
  <field name="domain">[('active', '=', True)]</field>
</record>
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Menus}
\begin{lstlisting}[style=xml]
<menuitem id="menu_school_root" name="School"/>
<menuitem id="menu_students"
		  name="Students"
		  parent="menu_school_root"
		  action="action_student"
		  sequence="10"/>
\end{lstlisting}
	\begin{itemize}
		\item \texttt{parent} builds the menu tree.
		\item \texttt{action} opens the window action.
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{Navigation from Python}
\begin{lstlisting}
def action_open_classroom(self):
	self.ensure_one()
	return {
		'type': 'ir.actions.act_window',
		'name': 'Classroom',
		'res_model': 'school.classroom',
		'view_mode': 'form',
		'res_id': self.classroom_id.id,
		'target': 'current',
	}
\end{lstlisting}
	\begin{itemize}
		\item \texttt{target}: \texttt{current}, \texttt{new}, \texttt{fullscreen}, etc.
		\item Buttons, smart buttons, and wizards all use this pattern.
	\end{itemize}
\end{frame}

\begin{frame}{Navigation Best Practices}
	\begin{itemize}
		\item Keep menu labels clear and short.
		\item Put the most common view mode first.
		\item Use domains/context in actions instead of hard-coding filters in every method.
		\item Reuse actions from smart buttons and wizards when possible.
	\end{itemize}
\end{frame}
