
\section{\textbf{\texttt{name\_search()}} ORM Method}

\begin{frame}[fragile]{\textbf{\texttt{name\_search()}} Method}
	\begin{itemize}
		\item The \textbf{\texttt{name\_search()}} method searches records by name / display name.
		\item It powers many2one dropdown search-as-you-type behavior.
		\item It returns a list of \texttt{(id, display\_name)} tuples.
		\item To customize search logic, developers often override \textbf{\texttt{\_name\_search()}}.
\begin{block}{Method Syntax}
\begin{lstlisting}
@api.model
def name_search(self, name='', args=None,
				operator='ilike', limit=100):
	return super().name_search(name=name, args=args,
		operator=operator, limit=limit)
\end{lstlisting}
\end{block}
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{Breaking Down the Method Signature}
	\begin{block}{}
		\begin{itemize}
			\item \textbf{\texttt{name}}: Text typed by the user.
			\item \textbf{\texttt{args}}: Extra domain restrictions.
			\item \textbf{\texttt{operator}}: Comparison operator, usually \texttt{ilike}.
			\item \textbf{\texttt{limit}}: Maximum number of suggestions.
\begin{lstlisting}
name = 'ahm'
args = [('active', '=', True)]
operator = 'ilike'
limit = 8
\end{lstlisting}
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame}[fragile]{What Does \texttt{name\_search()} Return?}
	\begin{block}{}
		\begin{itemize}
			\item It returns a list of tuples: \texttt{(id, display\_name)}.
\begin{lstlisting}
result = self.env['student.student'].name_search('ahm')
print(result)
\end{lstlisting}
			Output is (example):
\begin{lstlisting}
[
	(15, 'Ahmed Mohamed'),
	(22, 'Ahmed Ali'),
]
\end{lstlisting}
			\item Perfect format for many2one selection lists.
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame}[fragile]{How Many2one Uses \texttt{name\_search()}}
	\begin{itemize}
		\item User opens a many2one field and types text.
		\item Odoo calls \texttt{name\_search()} with that text.
		\item Matching records appear in the dropdown.
\begin{lstlisting}
self.env['student.student'].name_search(
	name='amina',
	args=[('active', '=', True)],
	operator='ilike',
	limit=10,
)
\end{lstlisting}
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{Searching Extra Fields}
	By default, name search focuses on the name / display name.
	You can extend it:
\begin{lstlisting}
@api.model
def _name_search(self, name='', args=None,
				operator='ilike', limit=100, order=None):
	args = args or []
	domain = ['|',
		('name', operator, name),
		('email', operator, name)]
	return self._search(domain + args, limit=limit, order=order)
\end{lstlisting}
	\begin{itemize}
		\item Now users can find a student by name or email.
		\item \texttt{\_name\_search()} is the private method commonly overridden.
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{Method Call}
\begin{block}{Method Call}

matches = self.env['student.student'].name_search(

\end{lstlisting}
	\begin{itemize}

	\end{itemize}
\end{block}
\end{frame}
